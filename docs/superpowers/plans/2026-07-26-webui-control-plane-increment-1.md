# Web UI control plane, increment 1: authenticated read-only panel — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the static `python3 -m http.server` web UI with a first-party
server that requires HTTP Basic auth on every path and exposes read-only JSON
endpoints, plus a dashboard page that consumes them.

**Architecture:** One new unprivileged Python 3 stdlib server
(`sbin/palwarden-webui`) serves the existing vendored editors *and* `/api/*`. Read
endpoints shell out to the existing read-only tools and return their JSON. No
mutating endpoints exist yet — `POST` stays unimplemented until increment 2 adds
the job queue and the root worker. Credentials live in a dedicated
`/etc/palworld/webui.env`, generated by a root-only `--init-credentials` mode.

**Tech Stack:** Python 3 standard library only (`http.server`, `subprocess`,
`hmac`, `secrets`, `base64`, `json`, `importlib`). Bash + `tests/lib/assert.sh`
for tests. systemd (bare metal) and s6-overlay (container) for supervision.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-07-26-webui-control-plane-design.md`.
- **Python 3 standard library only.** No new runtime dependencies; the image
  already installs `python3`.
- **New scripts carry SPDX headers:** `# SPDX-License-Identifier: AGPL-3.0-or-later`
  and `# SPDX-FileCopyrightText: 2026 Brian Grant`.
- **Host-portable:** never hardcode container assumptions. Every path is an env
  override with a bare-metal-preserving default (`CLAUDE.md` convention).
- **Do not modify** `webui/PalWorldSettingsEditor.html` or
  `webui/EngineIniPerformanceEditor.html`. They are vendored MIT; keep them
  byte-identical.
- **Fail closed:** refuse to start without credentials; refuse to start as root.
- **Never log credentials** — no `Authorization` header values in any log line.
- **Loopback default:** `PALWARDEN_WEBUI_BIND` defaults to `127.0.0.1`. The
  container overrides it to `0.0.0.0` because compose publishes to `127.0.0.1`.
- `PALWORLD_BACKUP_DIR` already means *config* backups
  (`/opt/palworld/config-backups`) in `palworld-config-apply-env` and
  `palworld-engine-config`. **Do not reuse that name.** World-save backups use
  the new `PALWARDEN_SAVE_BACKUP_DIR` (`/opt/palworld/backups`).
- Every task ends green: `./tests/lint.sh` and `./tests/run.sh` both pass.
- Commit messages: imperative subject, and no `Co-Authored-By` line is required
  by this plan (the repo's convention adds one; follow whatever the repo does).

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: AGPL-3.0-or-later
# SPDX-FileCopyrightText: 2026 Brian Grant
"""palwarden web UI server: static editors plus a read-only JSON API.

Runs UNPRIVILEGED on purpose. It executes only read-only tools; anything that
mutates the server is enqueued for the root worker (increment 2), so a bug in
this HTTP parser cannot restart the game or run SteamCMD.

Every path — including the static editors — requires HTTP Basic auth. Basic over
plain HTTP is acceptable only because the listener is loopback-bound and remote
access goes through an SSH tunnel, which supplies the encryption.
"""

from __future__ import annotations

import argparse
import os
import secrets
import sys
from pathlib import Path

WEBUI_ENV = Path(os.environ.get("PALWARDEN_WEBUI_ENV", "/etc/palworld/webui.env"))

REQUIRED_KEYS = ("WEBUI_USER", "WEBUI_PASSWORD", "WEBUI_TOKEN")


def load_credentials(path: Path = WEBUI_ENV) -> dict[str, str]:
    """Read KEY="value" lines. Exits with a clear message if unusable."""
    if not path.is_file():
        sys.exit(
            f"error: no web UI credentials at {path}. "
            "Run 'palwarden-webui --init-credentials' as root to create them."
        )
    values: dict[str, str] = {}
    for raw in path.read_text(errors="replace").splitlines():
        line = raw.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, _, val = line.partition("=")
        val = val.strip()
        if len(val) >= 2 and val[0] == val[-1] and val[0] in "\"'":
            val = val[1:-1]
        values[key.strip()] = val
    missing = [k for k in REQUIRED_KEYS if not values.get(k)]
    if missing:
        sys.exit(f"error: web UI credentials in {path} are missing: {', '.join(missing)}")
    return values


def init_credentials(path: Path = WEBUI_ENV) -> tuple[dict[str, str], bool]:
    """Create the credential file if absent. Never overwrites an existing one."""
    if path.is_file():
        return load_credentials(path), False
    creds = {
        "WEBUI_USER": os.environ.get("WEBUI_USER") or "admin",
        "WEBUI_PASSWORD": os.environ.get("WEBUI_PASSWORD") or secrets.token_urlsafe(24),
        "WEBUI_TOKEN": os.environ.get("WEBUI_TOKEN") or secrets.token_urlsafe(32),
    }
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + ".tmp")
    tmp.write_text("".join(f'{k}="{v}"\n' for k, v in creds.items()))
    tmp.chmod(0o600)
    tmp.replace(path)
    return creds, True


def main() -> int:
    ap = argparse.ArgumentParser(description="palwarden web UI server")
    ap.add_argument("--init-credentials", action="store_true",
                    help="create /etc/palworld/webui.env if absent, then exit")
    ap.add_argument("--serve", action="store_true", help="run the HTTP server")
    args = ap.parse_args()

    if args.init_credentials:
        creds, created = init_credentials(WEBUI_ENV)
        if created:
            print(f"Created {WEBUI_ENV} with new web UI credentials:")
            for key in REQUIRED_KEYS:
                print(f"  {key}={creds[key]}")
            print("Store these now; they are not printed again.")
        else:
            print(f"{WEBUI_ENV} already exists; leaving it unchanged.")
        return 0

    if args.serve:
        load_credentials(WEBUI_ENV)   # fail closed before binding anything
        print("serve not implemented yet", file=sys.stderr)
        return 0

    ap.print_help()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def check_basic(header: str | None, creds: dict[str, str]) -> bool:
    """Validate an HTTP Basic header in constant time."""
    if not header or not header.startswith("Basic "):
        return False
    try:
        decoded = base64.b64decode(header[6:], validate=True).decode("utf-8", "replace")
    except Exception:
        return False
    user, sep, password = decoded.partition(":")
    if not sep:
        return False
    ok_user = hmac.compare_digest(user, creds["WEBUI_USER"])
    ok_pass = hmac.compare_digest(password, creds["WEBUI_PASSWORD"])
    return ok_user and ok_pass


def allowed_roots() -> list[Path]:
    """Directories the static server may serve from.

    The web root contains a deliberate `current` symlink into the live config
    directory, so that directory is allowed too — but nothing else is.
    """
    roots = []
    for candidate in (WEB_ROOT, CONFIG_FILE.parent):
        try:
            roots.append(candidate.resolve())
        except OSError:
            continue
    return roots


def resolve_static(url_path: str) -> Path | None:
    """Map a URL path to a file inside the allowed roots, or None."""
    clean = urllib.parse.unquote(urllib.parse.urlparse(url_path).path)
    if clean in ("", "/"):
        clean = "/palwarden.html"
    candidate = (WEB_ROOT / clean.lstrip("/"))
    try:
        resolved = candidate.resolve()
    except OSError:
        return None
    if not any(resolved == root or root in resolved.parents for root in allowed_roots()):
        return None
    return resolved if resolved.is_file() else None
```

```python
def run_tool(argv: list[str], timeout: int = 30) -> dict:
    """Run a read-only tool. A failure is data, not an exception."""
    try:
        proc = subprocess.run(argv, text=True, capture_output=True,
                              timeout=timeout, check=False)
    except FileNotFoundError:
        return {"ok": False, "error": f"{argv[0]} is not installed"}
    except subprocess.TimeoutExpired:
        return {"ok": False, "error": f"{argv[0]} timed out after {timeout}s"}
    except OSError as exc:
        return {"ok": False, "error": f"{argv[0]} failed: {exc}"}
    return {
        "ok": True,
        "returncode": proc.returncode,
        "stdout": proc.stdout,
        "stderr": proc.stderr[:500],
    }


def run_tool_json(argv: list[str], timeout: int = 30) -> dict:
    result = run_tool(argv, timeout)
    if not result["ok"]:
        return result
    try:
        return {"ok": True, "data": json.loads(result["stdout"] or "{}")}
    except json.JSONDecodeError as exc:
        return {"ok": False, "error": f"{argv[0]} returned invalid JSON: {exc}"}


def pick_window(query: dict[str, list[str]], default: str = "24h",
                allowed: set[str] | None = None) -> str:
    """Validate a window parameter; never pass an unvetted string to a tool."""
    value = (query.get("window") or query.get("since") or [default])[0]
    if allowed is not None:
        return value if value in allowed else default
    return value if WINDOW_RE.match(value) else default


def load_parser_module():
    """Reuse the config parser's OptionSettings splitter instead of duplicating it.

    palworld-health-report already loads palworld-fps this way.

    try:
        text = CONFIG_FILE.read_text(errors="replace")
    except OSError as exc:
        return {"ok": False, "error": f"cannot read {CONFIG_FILE}: {exc}"}
    match = re.search(r"^OptionSettings=\((.*)\)[ \t]*$", text, re.MULTILINE)
    if not match:
        return {"ok": False, "error": f"no OptionSettings=(...) line in {CONFIG_FILE}"}
    try:
        split = load_parser_module().split_settings
    except Exception as exc:
        return {"ok": False, "error": f"cannot load the config parser: {exc}"}
    settings: dict[str, str] = {}
    for part in split(match.group(1)):
        key, sep, value = part.partition("=")
        if not sep:
            continue
        key = key.strip()
        value = value.strip()
        if key.lower() in SECRET_KEYS:
            settings[key] = "<redacted>"
        else:
            settings[key] = value[1:-1] if len(value) >= 2 and value[0] == value[-1] == '"' else value
    return {"ok": True, "data": settings}


def _list_dir(path: Path, limit: int = 50) -> dict:
    if not path.is_dir():
        return {"ok": True, "data": []}
    entries = []
    for item in sorted(path.iterdir(), key=lambda p: p.name, reverse=True)[:limit]:
        try:
            stat = item.stat()
        except OSError:
            continue
        entries.append({"name": item.name, "bytes": stat.st_size,
                        "modified": int(stat.st_mtime)})
    return {"ok": True, "data": entries}
```

```python
class Handler(http.server.BaseHTTPRequestHandler):
    server_version = "palwarden-webui"
    credentials: dict[str, str] = {}

    # --- helpers ----------------------------------------------------------
    def _send_json(self, status: int, payload: dict) -> None:
        body = json.dumps(payload, indent=2, sort_keys=True).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(body)

    def _challenge(self) -> None:
        body = b"authentication required\n"
        self.send_response(401)
        self.send_header("WWW-Authenticate", f'Basic realm="{REALM}"')
        self.send_header("Content-Type", "text/plain")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _authenticated(self) -> bool:
        if check_basic(self.headers.get("Authorization"), self.credentials):
            return True
        self._challenge()
        return False

    # --- routes -----------------------------------------------------------
    def do_GET(self) -> None:  # noqa: N802 - stdlib naming
        if not self._authenticated():
            return
        parsed = urllib.parse.urlparse(self.path)
        if parsed.path.startswith("/api/"):
            handler = READ_ENDPOINTS.get(parsed.path)
            if handler is None:
                self._send_json(404, {"ok": False, "error": "unknown endpoint"})
                return
            query = urllib.parse.parse_qs(parsed.query)
            try:
                self._send_json(200, handler(query))
            except Exception as exc:  # never leak a traceback to the browser
                self._send_json(500, {"ok": False, "error": f"internal error: {exc}"})
            return

        target = resolve_static(self.path)
        if target is None:
            self.send_error(404, "not found")
            return
        try:
            data = target.read_bytes()
        except OSError:
            self.send_error(404, "not found")
            return
        ctype = {
            ".html": "text/html; charset=utf-8",
            ".ini": "text/plain; charset=utf-8",
            ".css": "text/css",
            ".js": "application/javascript",
            ".svg": "image/svg+xml",
            ".json": "application/json",
        }.get(target.suffix, "application/octet-stream")
        self.send_response(200)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(data)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(data)

    def do_POST(self) -> None:  # noqa: N802
        if not self._authenticated():
            return
        # Mutating endpoints arrive in increment 2, together with the root worker.
        self._send_json(501, {"ok": False,
                              "error": "mutating endpoints are not implemented yet"})

    # --- logging: request lines only, never headers -----------------------
    def log_message(self, fmt: str, *args) -> None:
        sys.stderr.write("%s %s\n" % (self.address_string(), fmt % args))


class ThreadingHTTPServer(socketserver.ThreadingMixIn, http.server.HTTPServer):
    daemon_threads = True
    allow_reuse_address = True


def make_server(creds: dict[str, str]) -> ThreadingHTTPServer:
    Handler.credentials = creds
    return ThreadingHTTPServer((BIND, PORT), Handler)
```
